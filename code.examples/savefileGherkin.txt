	P(1) => line
	line=line.replace("-"," ").trim
	line=="#"

# Process a list of table lines, interactive if no param
# --
	P(1) => lines

	interactive=false
	if (lines==null) {
		println("Paste table data and end with .")
		lines=readLines(".")
		interactive=true
	}

	error(Sys.getType(lines) != "List", "Expected list of lines")

	lines->line out("-"+line) | _=>lines  # prefix in order to handle comments
	
	# max length for columns
	maxLengths=Std.Data.yes(50,0)  # never more than 50 columns

	colCount=0
	lines->line
		Inner {
			col=0
			reject(IsLine(line) || line.trim=="" || !line.contains("|"))  # ignore these now
			line.trim.split("|")->part
				length = part.trim.length
				if (length > maxLengths.get(col))
					maxLengths.set(col,length)
			col=col+1
			if (col>colCount) colCount=col
		}
	|


	# Find indent from first line
	indent=" ".times(lines.first.before("|").length-1)
	commentIndent=" ".times(indent.length-1)

	# Format lines 
	lines->line
		resultLine=""
		if (IsLine(line)) {
			# generate separator string
			n=maxLengths.sum - maxLengths.get(0) + 2 - 1 + (colCount-1) * 3 - 2
			
			resultLine=indent + "# " + "-".times(n)
		} else if (line.trim=="" || line.trim=="-") {
			resultLine=""
		} else if (!line.contains("|")) {
			resultLine=line
		} else Inner {
			col=0
			line.trim.split("|")->part
				if (col==0) {
					resultLine = resultLine + if (part.trim=="-", indent + "| ", "#"+commentIndent+"| ")
				} else {
					maxLength=maxLengths.get(col)
					if (part.trim.startsWith("#")) {
						resultLine=resultLine + part.trim
					} else {
						s=part.trim + " ".times(maxLength)
						resultLine=resultLine + s.first(maxLength) + " | "
					}
				}
				col=col+1
		}
		if (interactive) println(resultLine) else out(resultLine)

/FormatTableLines


FormatTableLines
/FTL


# Fix table data in Gherkin manuscripts - first line determines indent
# --
	P(1) as File => file

	tableData=List
	result=List

	file.read(true)->line			# detab
		isTableData=line.trim.startsWith("|")
			|| (line.trim.startsWith("#") && line.after("#").trim.startsWith("|"))
			|| (!tableData.empty && IsLine(line))
			|| (!tableData.empty && line.trim=="")

		if (isTableData) {
			tableData.add(line)
			continue
		} else {
			if (!tableData.empty) {
				result=result + FormatTableLines(tableData)
				tableData=List
			}
			result.add(line)
		}
		if (!tableData.empty) result=result + FormatTableLines(tableData)
	|
	if (!tableData.empty) result=result + FormatTableLines(tableData)
	file.create(result)
/FixFileTables


# Generate doc from python Gherkin/Behave file
# --
	P(1) as File => pythonFile
	
	count=0  # distance since @-line
	pythonFile.read->line
		if (line.startsWith("@")) {
			out("")
			out(line)
			count=0
			continue
		}
		count=count+1
		if (line.trim.startsWith("#") && line.contains("|") && count <= 2)
			out(line)
	| _.print
