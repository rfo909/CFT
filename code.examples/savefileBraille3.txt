	=>lines
	symbols=List
	codes=List
	lines->line
		reject(line.trim=="")
		line.split->x
			assert(x.contains(":"))
			
			symbol=x.before(":")
			code=x.after(":")

			error(symbols.contains(symbol),"Duplicate symbol " + symbol)
			error(codes.contains(code), "Duplicate code " + code)
			
			symbols.add(symbol)
			codes.add(code)
			
			out(x)
	| _.sort
/Data
